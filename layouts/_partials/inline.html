{{- /* A piece of markdown as HTML the way the site sets it: Czech typography, italics as , source links from [1] and non-breaking year ranges. Takes a dict with the keys page, text and an optional block (for multi-paragraph text). */ -}} {{- $options := dict "display" "inline" -}} {{- if .block }}{{ $options = dict }}{{ end -}} {{- $html := .page.RenderString $options (partial "typography.html" .text) -}} {{- $html = replaceRE `(.*?)` "${1}" $html -}} {{- /* A link out of the site opens in a new tab: the card being learned stays where it was. Links within the site — to an answer, to a source in the list below — keep the tab, they are the reading itself. */ -}} {{- $html = replaceRE `${1}` $html -}} {{- $html = replaceRE `,${1}` $html -}} {{- /* A dental formula is one fraction: the teeth of the upper jaw over those of the lower. Written in markdown as "3 1 4 2 / 3 1 4 3", it comes out stacked with a rule between the two rows. */ -}} {{- $html = replaceRE `(\d \d \d \d) / (\d \d \d \d)` `${1}${2}` $html -}} {{- return (safeHTML $html) -}}
